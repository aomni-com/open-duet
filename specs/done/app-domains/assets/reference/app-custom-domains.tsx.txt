'use client'

import {
  ArrowClockwiseIcon,
  CheckCircleIcon,
  GlobeIcon,
  PlusIcon,
  TrashIcon,
  WarningCircleIcon,
} from '@phosphor-icons/react'
import { useAction, useQuery } from 'convex/react'
import { useTranslations } from 'next-intl'
import { useState } from 'react'

import { api } from '@repo/backend/convex/_generated/api'
import { type Doc, type Id } from '@repo/backend/convex/_generated/dataModel'
import { Badge } from '@repo/ui/components/shadcn/badge'
import { Button } from '@repo/ui/components/shadcn/button'
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
  DialogTrigger,
} from '@repo/ui/components/shadcn/dialog'
import { Input } from '@repo/ui/components/shadcn/input'
import { Label } from '@repo/ui/components/shadcn/label'
import { Skeleton } from '@repo/ui/components/shadcn/skeleton'
import { toast } from '@repo/ui/components/shadcn/sonner'

type CustomDomain = Doc<'customDomains'>

function StatusBadge({ status }: { status: CustomDomain['status'] }) {
  const t = useTranslations('app.apps.domains')
  switch (status) {
    case 'active':
      return (
        <Badge
          variant="outline"
          className="border-emerald-500/30 bg-emerald-500/10 text-[10px] text-emerald-600"
        >
          <CheckCircleIcon className="mr-0.5 size-3" />
          {t('statusActive')}
        </Badge>
      )
    case 'verifying':
      return (
        <Badge
          variant="outline"
          className="border-amber-500/30 bg-amber-500/10 text-[10px] text-amber-600"
        >
          {t('statusVerifying')}
        </Badge>
      )
    case 'pending':
      return (
        <Badge variant="outline" className="text-[10px]">
          {t('statusPending')}
        </Badge>
      )
    case 'failed':
      return (
        <Badge
          variant="outline"
          className="border-red-500/30 bg-red-500/10 text-[10px] text-red-600"
        >
          <WarningCircleIcon className="mr-0.5 size-3" />
          {t('statusFailed')}
        </Badge>
      )
  }
}

function VerificationRecords({ domain }: { domain: CustomDomain }) {
  const t = useTranslations('app.apps.domains')
  const hasVerification = domain.verification && domain.verification.length > 0
  const hasConflicts = domain.conflicts && domain.conflicts.length > 0
  if (!hasVerification && !hasConflicts) return null

  return (
    <div className="mt-3 space-y-3">
      {hasVerification ? (
        <div className="rounded-md border bg-muted/30 p-3 space-y-2">
          <p className="text-xs font-medium">{t('dnsRequired')}</p>
          {domain.verification!.map((record, idx) => (
            <div key={idx} className="grid grid-cols-[80px_1fr] gap-2 text-xs font-mono">
              <span className="text-muted-foreground uppercase">{record.type}</span>
              <div>
                <div className="break-all">{record.domain}</div>
                <div className="break-all text-muted-foreground">{record.value}</div>
                {record.reason ? (
                  <div className="mt-0.5 break-all text-[10px] text-muted-foreground/70 font-sans italic">
                    {record.reason}
                  </div>
                ) : null}
              </div>
            </div>
          ))}
          <p className="text-xs text-muted-foreground pt-1">{t('dnsHint')}</p>
        </div>
      ) : null}

      {hasConflicts ? (
        <div className="rounded-md border border-red-500/30 bg-red-500/5 p-3 space-y-2">
          <p className="text-xs font-medium text-red-600">{t('conflicts')}</p>
          {domain.conflicts!.map((record, idx) => (
            <div key={idx} className="grid grid-cols-[80px_1fr] gap-2 text-xs font-mono">
              <span className="text-muted-foreground uppercase">{record.type}</span>
              <div>
                <div className="break-all">{record.name}</div>
                <div className="break-all text-muted-foreground">{record.value}</div>
              </div>
            </div>
          ))}
          <p className="text-xs text-muted-foreground pt-1">{t('conflictsHint')}</p>
        </div>
      ) : null}
    </div>
  )
}

export function AddDomainDialog({
  organizationId,
  appSlug,
  onAdded,
  trigger,
}: {
  organizationId: Id<'organizations'>
  appSlug: string
  onAdded?: () => void
  trigger?: React.ReactNode
}) {
  const t = useTranslations('app.apps.domains')
  const [open, setOpen] = useState(false)
  const [value, setValue] = useState('')
  const [submitting, setSubmitting] = useState(false)
  const addDomain = useAction(api.domains.customDomains.actions.addDomain)

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault()
    const trimmed = value.trim()
    if (!trimmed) return
    setSubmitting(true)
    try {
      await addDomain({ organizationId, appSlug, domain: trimmed })
      toast.success(t('addedToast', { domain: trimmed }))
      setOpen(false)
      setValue('')
      onAdded?.()
    } catch (error) {
      const message = error instanceof Error ? error.message : t('addError')
      toast.error(message)
    } finally {
      setSubmitting(false)
    }
  }

  return (
    <Dialog open={open} onOpenChange={setOpen}>
      <DialogTrigger asChild>
        {trigger ?? (
          <Button variant="outline" size="sm" className="gap-2">
            <PlusIcon className="size-3.5" />
            {t('addDomain')}
          </Button>
        )}
      </DialogTrigger>
      <DialogContent>
        <form onSubmit={handleSubmit}>
          <DialogHeader>
            <DialogTitle>{t('addDomainTitle')}</DialogTitle>
            <DialogDescription>{t('addDomainDescription')}</DialogDescription>
          </DialogHeader>
          <div className="py-4 space-y-2">
            <Label htmlFor="domain-input">{t('domainLabel')}</Label>
            <Input
              id="domain-input"
              placeholder={t('domainPlaceholder')}
              value={value}
              onChange={e => setValue(e.target.value)}
              autoFocus
            />
          </div>
          <DialogFooter>
            <Button type="button" variant="outline" onClick={() => setOpen(false)}>
              {t('cancel')}
            </Button>
            <Button type="submit" disabled={submitting || !value.trim()}>
              {submitting ? t('adding') : t('addDomain')}
            </Button>
          </DialogFooter>
        </form>
      </DialogContent>
    </Dialog>
  )
}

export function DomainRow({ domain }: { domain: CustomDomain }) {
  const t = useTranslations('app.apps.domains')
  const removeDomain = useAction(api.domains.customDomains.actions.removeDomain)
  const refreshDomain = useAction(api.domains.customDomains.actions.refreshDomain)
  const [busy, setBusy] = useState(false)

  const handleRefresh = async () => {
    setBusy(true)
    try {
      const { status } = await refreshDomain({ domainId: domain._id })
      toast.success(status === 'active' ? t('refreshVerified') : t('refreshStillVerifying'))
    } catch (error) {
      const message = error instanceof Error ? error.message : t('refreshError')
      toast.error(message)
    } finally {
      setBusy(false)
    }
  }

  const handleRemove = async () => {
    if (!confirm(t('removeConfirm', { domain: domain.domain }))) return
    setBusy(true)
    try {
      await removeDomain({ domainId: domain._id })
      toast.success(t('removedToast', { domain: domain.domain }))
    } catch (error) {
      const message = error instanceof Error ? error.message : t('removeError')
      toast.error(message)
    } finally {
      setBusy(false)
    }
  }

  return (
    <div className="rounded-lg border p-4">
      <div className="flex items-start justify-between gap-4">
        <div className="min-w-0 flex-1">
          <div className="flex items-center gap-2">
            <GlobeIcon className="size-4 text-muted-foreground shrink-0" />
            <a
              href={`https://${domain.domain}`}
              target="_blank"
              rel="noopener noreferrer"
              className="truncate text-sm font-medium hover:underline"
            >
              {domain.domain}
            </a>
            <StatusBadge status={domain.status} />
          </div>
          {domain.lastError ? (
            <p className="mt-1 text-xs text-red-600">{domain.lastError}</p>
          ) : null}
        </div>
        <div className="flex items-center gap-1">
          <Button
            variant="ghost"
            size="sm"
            onClick={handleRefresh}
            disabled={busy}
            title={t('refreshTitle')}
          >
            <ArrowClockwiseIcon className="size-3.5" />
          </Button>
          <Button
            variant="ghost"
            size="sm"
            onClick={handleRemove}
            disabled={busy}
            className="text-destructive hover:text-destructive"
            title={t('removeTitle')}
          >
            <TrashIcon className="size-3.5" />
          </Button>
        </div>
      </div>
      {domain.status !== 'active' ? <VerificationRecords domain={domain} /> : null}
    </div>
  )
}

/**
 * Full management UI for a single app's custom domains.
 * Used by the per-app settings surface and by the org-level Custom Domains tab.
 */
export function AppCustomDomains({
  organizationId,
  appSlug,
  emptyHint,
}: {
  organizationId: Id<'organizations'>
  appSlug: string
  emptyHint?: string
}) {
  const t = useTranslations('app.apps.domains')
  const domains = useQuery(api.domains.customDomains.queries.listForApp, {
    organizationId,
    appSlug,
  })

  if (domains === undefined) {
    return (
      <div className="space-y-2">
        <Skeleton className="h-16 w-full" />
        <Skeleton className="h-16 w-full" />
      </div>
    )
  }

  return (
    <div className="space-y-4">
      <div className="flex items-center justify-between">
        <div>
          <h3 className="text-sm font-semibold">{t('sectionTitle')}</h3>
          <p className="text-xs text-muted-foreground">{t('sectionDescription')}</p>
        </div>
        <AddDomainDialog organizationId={organizationId} appSlug={appSlug} />
      </div>
      {domains.length === 0 ? (
        <p className="text-xs text-muted-foreground">{emptyHint ?? t('emptyHint')}</p>
      ) : (
        <div className="space-y-2">
          {domains.map(d => (
            <DomainRow key={d._id} domain={d} />
          ))}
        </div>
      )}
    </div>
  )
}
